# 3c. CPU Family & SoC Resolution
detect_dynamic_cpu() {
    # 1. Linux generic SoC bus (Snapdragon, Exynos, etc.)
    for soc_dir in /sys/devices/soc0 /sys/bus/soc/devices/soc0; do
        if [ -f "$soc_dir/machine" ]; then
            local soc_mach="$(tr -d '\0\r\n' < "$soc_dir/machine" 2>/dev/null)"
            if [ -n "$soc_mach" ]; then echo "$soc_mach"; return; fi
        fi
    done

    # 2. Positional Devicetree SoC extraction
    local dt_comp="/proc/device-tree/compatible"
    [ ! -f "$dt_comp" ] && dt_comp="/sys/firmware/devicetree/base/compatible"
    if [ -f "$dt_comp" ]; then
        local soc_line="$(tr '\0' '\n' < "$dt_comp" 2>/dev/null | sed -n '2p')"
        [ -z "$soc_line" ] && soc_line="$(tr '\0' '\n' < "$dt_comp" 2>/dev/null | head -n1)"
        if [ -n "$soc_line" ]; then
            local clean_soc="${soc_line#*,}"
            clean_soc="${clean_soc#sun*-}"
            # Normalize Allwinner BSP kernel silicon codes to standard PortMaster names
            case "$clean_soc" in
                sun50iw9*)  clean_soc="h700" ;;
                sun50iw10*) clean_soc="a133plus" ;;
                sun50iw12*) clean_soc="a527" ;;
                sun50iw6*)  clean_soc="h6" ;;
            esac
            if [ -n "$clean_soc" ]; then echo "$clean_soc"; return; fi
        fi
    fi

    # 3. /proc/cpuinfo Hardware or model name fallback
    if [ -f "/proc/cpuinfo" ]; then
        local cpu_hw="$(grep -iE '^(model name|Hardware)' /proc/cpuinfo 2>/dev/null | head -n1 | cut -d: -f2- | sed 's/^[ \t]*//' | tr -d '\r\n')"
        if [ -n "$cpu_hw" ] && [ "$cpu_hw" != "Unknown" ] && [ "$cpu_hw" != "dummy" ]; then
            echo "$cpu_hw"; return
        fi
    fi

    uname -m 2>/dev/null || echo "Unknown"
}
export DEVICE_CPU="$(detect_dynamic_cpu)"

# 3d. RAM calculation in GB (Ceiled to actual physical capacity)

    MEM_TOTAL_KB="$(awk '/^(MemTotal|CmaTotal):/ {sum += $2} END {print sum}' /proc/meminfo 2>/dev/null)"
    if [ -n "$MEM_TOTAL_KB" ] && [ "$MEM_TOTAL_KB" -gt 0 ]; then
        # Add ~12.5% (+ KB/8) + 64MB buffer to compensate for kernel/firmware overhead
        export DEVICE_RAM=$(( (MEM_TOTAL_KB + (MEM_TOTAL_KB / 8) + 65536) / 1048576 ))

        # 0 GB -> 512 MB | 1-4 GB -> rounded MB | > 4 GB -> actual MB
        export DEVICE_RAM_MB=$(( DEVICE_RAM > 4 ? (MEM_TOTAL_KB + 1023) / 1024 : (DEVICE_RAM ? DEVICE_RAM * 1024 : 512) ))
    fi
fi

# Dynamic GCD Aspect Ratio calculation
calc_gcd() {
    local a=$1 b=$2
    while [ "$b" -ne 0 ]; do
        local t=$b
        b=$((a % b))
        a=$t
    done
    echo "$a"
}

GCD=$(calc_gcd "$DISPLAY_WIDTH" "$DISPLAY_HEIGHT")
if [ "$GCD" -gt 0 ] 2>/dev/null; then
    export ASPECT_X=$(( DISPLAY_WIDTH / GCD ))
    export ASPECT_Y=$(( DISPLAY_HEIGHT / GCD ))
    if [ "$ASPECT_X" -eq 8 ] && [ "$ASPECT_Y" -eq 5 ]; then
        export ASPECT_X=16
        export ASPECT_Y=10
    fi
else
    export ASPECT_X=4
    export ASPECT_Y=3
fi

# Helper 1: Universal Big-Endian DTS integer reader (od -> hexdump -> xxd)
_read_dts_be32() {
    _hex=""
    if command -v od >/dev/null 2>&1; then
        _hex=$(od -An -N4 -tx1 "$1" 2>/dev/null | tr -d ' \n')
    elif command -v hexdump >/dev/null 2>&1; then
        _hex=$(hexdump -n4 -e '1/1 "%02x"' "$1" 2>/dev/null)
    elif command -v xxd >/dev/null 2>&1; then
        _hex=$(xxd -p -l 4 "$1" 2>/dev/null)
    fi
    echo $(( 0x${_hex:-0} ))
}

# Helper 2: POSIX zero-fork line reader
_read_line() {
    [ -r "$1" ] || return 1
    _val=""
    read -r _val < "$1" 2>/dev/null
    echo "$_val"
}

# Helper 3: Validates if an IIO SARADC channel is a real stick or floating/pegged
_is_iio_channel_active() {
    local ch="$1"
    local raw_file="/sys/bus/iio/devices/iio:device0/in_voltage${ch}_raw"
    [ -r "$raw_file" ] || return 0

    local raw_val="$(_read_line "$raw_file")"
    raw_val=$(( raw_val + 0 ))

    # Saturated rail (>= 1020 on 10-bit / >= 4090 on 12-bit) or 0 indicates unconnected/floating line
    if [ "$raw_val" -ge 1020 ] || [ "$raw_val" -le 5 ]; then
        return 1 # Inactive / Disconnected
    fi
    return 0 # Real connected analog stick
}

detect_dynamic_controls() {
    local has_physical_dt_joypad=0
    local dt_joypad_sticks=0
    local dt_joypad_triggers="N"

    # 1. Authoritative Device Tree Hardware Scan (ARM SBCs)
    for dt_node in /sys/firmware/devicetree/base/*joypad* /sys/firmware/devicetree/base/*/*joypad* \
                   /sys/firmware/devicetree/base/*joystick* /sys/firmware/devicetree/base/*/*joystick* \
                   /sys/class/input/input*/device/of_node; do
        [ -d "$dt_node" ] || continue

        # Strict guard: Must contain ADC joystick properties (filters out power keys and audio codecs)
        [ -f "$dt_node/io-channels" ] || [ -f "$dt_node/amux-count" ] || [ -f "$dt_node/direct-adc" ] || continue

        has_physical_dt_joypad=1

        # Case A: direct-adc is set -> amux is ignored by driver
        if [ -f "$dt_node/direct-adc" ] && [ -f "$dt_node/io-channels" ]; then
            local io_sz=$(wc -c < "$dt_node/io-channels" 2>/dev/null || echo 0)
            local num_channels=$(( io_sz / 8 ))
            if [ "$num_channels" -ge 4 ]; then
                dt_joypad_sticks=2
            elif [ "$num_channels" -ge 2 ]; then
                dt_joypad_sticks=1
            fi
        fi

        # Case B: amux-count (Multiplexed ADC)
        if [ "$dt_joypad_sticks" -eq 0 ] && [ -f "$dt_node/amux-count" ]; then
            local amux_cnt=$(_read_dts_be32 "$dt_node/amux-count")
            if [ "$amux_cnt" -ge 6 ]; then
                dt_joypad_sticks=2; dt_joypad_triggers="Y"
            elif [ "$amux_cnt" -ge 4 ]; then
                dt_joypad_sticks=2
            elif [ "$amux_cnt" -ge 2 ]; then
                dt_joypad_sticks=1
            fi
        fi

        # Case C: io-channels + SARADC Live Voltage Verification
        if [ "$dt_joypad_sticks" -eq 0 ] && [ -f "$dt_node/io-channels" ]; then
            local io_sz=$(wc -c < "$dt_node/io-channels" 2>/dev/null || echo 0)
            local num_channels=$(( io_sz / 8 ))
            if [ "$num_channels" -ge 4 ]; then
                if _is_iio_channel_active 0 && _is_iio_channel_active 1; then
                    dt_joypad_sticks=2
                    [ "$num_channels" -ge 6 ] && dt_joypad_triggers="Y"
                else
                    dt_joypad_sticks=1
                fi
            elif [ "$num_channels" -ge 2 ]; then
                dt_joypad_sticks=1
            fi
        fi

        [ "$dt_joypad_sticks" -gt 0 ] && break
    done

    # 2. Input Subsystem Telemetry: Physical Hardware vs Virtual uinput Daemons
    local has_physical_hw_stick=0
    local has_physical_hw_trig="N"
    local virtual_claimed_sticks=0
    local virtual_claimed_trig="N"

    for input_dir in /sys/class/input/input*; do
        [ -d "$input_dir" ] || continue
        local dev_name="$(_read_line "$input_dir/name")"
        local dev_props="$(_read_line "$input_dir/properties")"
        local dev_abs="$(_read_line "$input_dir/capabilities/abs")"

        # Skip devices without absolute axes
        if [ -z "$dev_abs" ] || [ "$dev_abs" = "0" ]; then
            continue
        fi

        # Filter Touchscreens & Direct Pointers (PROP_DIRECT=1, PROP_POINTER=2, PROP_SEMI_MT=8)
        [ -n "$dev_props" ] && [ $(( 0x${dev_props:-0} & 0xb )) -ne 0 ] && continue

        # Filter Multi-Touch bitmasks (screens / digitizers)
        case "$dev_abs" in ??????????*) continue ;; esac

        local lowest="${dev_abs##* }"
        lowest="${lowest##*,}"
        local val=$(( 0x${lowest:-0} ))

        # Filter Touch/Stylus/Digitizers via ABS_PRESSURE (bit 24) or ABS_DISTANCE (bit 25)
        [ $(( val & 0x3000000 )) -ne 0 ] && continue

        # Filter isolated 2D pointers that have ONLY ABS_X/Y and no other gamepad axes
        if [ $(( val & 0x3 )) -eq 3 ] && [ $(( val & 0x3063c )) -eq 0 ]; then
            continue
        fi

        # Classify bus: Physical Hardware vs Virtual (uinput daemon)
        local dev_path=""
        [ -e "$input_dir/device" ] && dev_path=$(readlink -f "$input_dir/device" 2>/dev/null)
        local is_virtual=0
        case "$dev_path" in
            */devices/virtual/*|"") is_virtual=1 ;;
        esac

        local node_sticks=0
        local node_trig="N"

        [ $(( val & 0x3 )) -eq 3 ] && node_sticks=1
        { [ $(( val & 0x18 )) -eq 24 ] || [ $(( val & 0x24 )) -eq 36 ]; } && node_sticks=2
        { [ $(( val & 0x600 )) -eq 1536 ] || [ $(( val & 0x3f )) -eq 63 ]; } && node_trig="Y"

        if [ "$is_virtual" -eq 0 ]; then
            [ "$node_sticks" -gt "$has_physical_hw_stick" ] && has_physical_hw_stick="$node_sticks"
            [ "$node_trig" = "Y" ] && has_physical_hw_trig="Y"
        else
            [ "$node_sticks" -gt "$virtual_claimed_sticks" ] && virtual_claimed_sticks="$node_sticks"
            [ "$node_trig" = "Y" ] && virtual_claimed_trig="Y"
        fi
    done

# 3. Dynamic Consensus Arbitration
    if [ "$dt_joypad_sticks" -gt 0 ]; then
        ANALOG_STICKS="$dt_joypad_sticks"
        ANALOG_TRIGGERS="$dt_joypad_triggers"
    elif [ "$has_physical_hw_stick" -ge 2 ]; then
        ANALOG_STICKS="$has_physical_hw_stick"
        ANALOG_TRIGGERS="$has_physical_hw_trig"
    elif [ "$virtual_claimed_sticks" -gt 0 ]; then
        # On x86/x86_64 devices (Steam Deck, ROG Ally, etc.), trust userspace daemons (hhd, Steam, etc.)
        if [ "$DEVICE_HAS_X86" = "Y" ] || [ "$DEVICE_HAS_X86_64" = "Y" ] || [ ! -d "/proc/device-tree" ]; then
            ANALOG_STICKS="$virtual_claimed_sticks"
            ANALOG_TRIGGERS="$virtual_claimed_trig"
        # On ARM platforms with DeviceTree: reject virtual stick claims if no physical hardware exists
        elif [ "$dt_joypad_sticks" -eq 0 ] && [ "$has_physical_hw_stick" -eq 0 ]; then
            ANALOG_STICKS=0
            ANALOG_TRIGGERS="N"
        else
            ANALOG_STICKS="$virtual_claimed_sticks"
            ANALOG_TRIGGERS="$virtual_claimed_trig"
        fi
    elif [ "$has_physical_hw_stick" -gt 0 ]; then
        ANALOG_STICKS="$has_physical_hw_stick"
        ANALOG_TRIGGERS="$has_physical_hw_trig"
    else
        ANALOG_STICKS=0
        ANALOG_TRIGGERS="N"
    fi

    export ANALOG_STICKS
    export ANALOG_TRIGGERS

}

detect_dynamic_controls

# ------------------------------------------------------------------------------
# 5b. CAPABILITIES & TAG GENERATION
# ------------------------------------------------------------------------------
CAPS=()

# 1. Architecture
CAPS+=("$DEVICE_ARCH")
[ "$DEVICE_HAS_ARMHF" = "Y" ] && CAPS+=("armhf")
[ "$DEVICE_HAS_AARCH64" = "Y" ] && CAPS+=("aarch64")
[ "$DEVICE_HAS_X86" = "Y" ] && CAPS+=("x86")
[ "$DEVICE_HAS_X86_64" = "Y" ] && CAPS+=("x86_64")

# 2. Dynamic OpenGL & Vulkan feature discovery
HAS_DESKTOP_GL="N"
for gl_path in /usr/lib*/libGL.so* /usr/lib/*/libGL.so* /lib*/libGL.so*; do
    if [ -e "$gl_path" ]; then HAS_DESKTOP_GL="Y"; break; fi
done
[ "$HAS_DESKTOP_GL" = "Y" ] && CAPS+=("opengl")

for vk_path in /usr/lib*/libvulkan.so* /usr/lib/*/libvulkan.so* /usr/share/vulkan/icd.d/*.json; do
    if [ -e "$vk_path" ]; then CAPS+=("vulkan"); break; fi
done

# 3. Hardware Power Tiers derived from Core Count, Clock Freq, and RAM
CPU_LOWER="$(echo "$DEVICE_CPU" | tr '[:upper:]' '[:lower:]')"

# 4a. 'power' - Enabled on all devices EXCEPT rk3326 and px30
case "$CPU_LOWER" in
    *rk3326*|*px30*)
        # Excluded: Low power legacy SoCs
        ;;
    *)
        CAPS+=("power")
        ;;
esac

# 4b. 'ultra' - Requires >= 4GB RAM AND excludes budget/low-power SoCs
if [ "$DEVICE_RAM" -ge 4 ]; then
    case "$CPU_LOWER" in
        *rk3326*|*h700*|*a133*|*a133plus*|*a527*|*px30*|*sun50iw9*|*sun50iw10*)
            # Excluded budget chipsets even if paired with higher memory
            ;;
        *)
            CAPS+=("ultra")
            ;;
    esac
fi

# 4. Display & Aspect Ratio
CAPS+=("${DISPLAY_WIDTH}x${DISPLAY_HEIGHT}")
CAPS+=("${ASPECT_X}:${ASPECT_Y}")
[ "$ASPECT_X" -eq 16 ] && [ "$ASPECT_Y" -eq 10 ] && CAPS+=("16:9")

if [ "$DISPLAY_WIDTH" -ge 960 ] || [ "$DISPLAY_HEIGHT" -ge 720 ]; then
    CAPS+=("hires")
elif [ "$DISPLAY_WIDTH" -lt 640 ] || [ "$DISPLAY_HEIGHT" -lt 480 ]; then
    CAPS+=("lowres")
fi
if [ "$ASPECT_X" -ge 16 ] || [ $(( ASPECT_X * 10 / ASPECT_Y )) -ge 15 ]; then
    CAPS+=("wide")
elif [ "$ASPECT_X" -eq "$ASPECT_Y" ]; then
    CAPS+=("square")
fi

# 5. Analog Controller Tags
for ((i=0; i<=$ANALOG_STICKS; i++)); do
    CAPS+=("analog_$i")
done
[ "$ANALOG_TRIGGERS" = "Y" ] && CAPS+=("analog_triggers")

# 6. Cumulative RAM tags
for tier in 1 2 4 8 16 32; do
    if [ "$DEVICE_RAM" -ge "$tier" ]; then CAPS+=("${tier}gb"); fi
done

# 7. Dynamic Restore Capability (Checks writable backup / save directory availability)
for sav_dir in /userdata/saves /storage/roms/ports /roms/ports /mnt/SDCARD/Roms/PORTS "$HOME/.local/share"; do
    if [ -d "$sav_dir" ] && [ -w "$sav_dir" ]; then
        CAPS+=("restore")
        break
    fi
done

# Deduplicate capabilities
export DEVICE_CAPABILITIES="$(echo "${CAPS[@]}" | tr ' ' '\n' | sort -u | tr '\n' ' ' | sed 's/[ \t]*$//')"

save_device_info() {
    local target_file="$1"
    cat <<EOF > "$target_file"
DEVICE_INFO_VERSION="$DEVICE_INFO_VERSION"
PM_VERSION="$PM_VERSION"
CFW_NAME="$CFW_NAME"
CFW_VERSION="$CFW_VERSION"
CFW_GLIBC="$CFW_GLIBC"
DEVICE_NAME="$DEVICE_NAME"
DEVICE_CPU="$DEVICE_CPU"
DEVICE_ARCH="$DEVICE_ARCH"
DEVICE_RAM="$DEVICE_RAM"
DEVICE_RAM_MB="$DEVICE_RAM_MB"
DEVICE_HAS_ARMHF="$DEVICE_HAS_ARMHF"
DEVICE_HAS_AARCH64="$DEVICE_HAS_AARCH64"
DEVICE_HAS_X86="$DEVICE_HAS_X86"
DEVICE_HAS_X86_64="$DEVICE_HAS_X86_64"
DISPLAY_WIDTH="$DISPLAY_WIDTH"
DISPLAY_HEIGHT="$DISPLAY_HEIGHT"
DISPLAY_ORIENTATION="$DISPLAY_ORIENTATION"
ASPECT_X="$ASPECT_X"
ASPECT_Y="$ASPECT_Y"
ANALOG_STICKS="$ANALOG_STICKS"
ANALOG_TRIGGERS="$ANALOG_TRIGGERS"
DEVICE_CAPABILITIES="$DEVICE_CAPABILITIES"
EOF
}

if [ -d "$CONTROL_DIR" ] && [ -w "$CONTROL_DIR" ]; then
    # Sanitize names for filesystem safety (lowercase, replace spaces with underscores)
    SAFE_CFW="$(echo "$CFW_NAME" | tr '[:upper:] ' '[:lower:]_')"
    SAFE_DEV="$(echo "$DEVICE_NAME" | tr '[:upper:] ' '[:lower:]_')"
    
    NAMED_ENV="$CONTROL_DIR/device_info_${SAFE_CFW}_${SAFE_DEV}.env"
    save_device_info "$NAMED_ENV"
    
    # Also write / link canonical device_info.env
    cp -f "$NAMED_ENV" "$CONTROL_DIR/device_info.env" 2>/dev/null
fi
