#!/bin/bash
# ==============================================================================
# PortMaster Device Information Provider (Dynamic Modernized v0.2.9)
# ==============================================================================

(return 0 2>/dev/null) && SOURCED=1 || SOURCED=0

if [ "$SOURCED" -eq 1 ] && [ -n "$DEVICE_INFO_VERSION" ]; then
    return 0 2>/dev/null
fi

export DEVICE_INFO_VERSION="0.2.9"

# ------------------------------------------------------------------------------
# 0. DIRECTORY SETUP
# ------------------------------------------------------------------------------
if [ -z "$controlfolder" ]; then
    if [ -n "$PORTMASTER_HOME" ] && [ -d "$PORTMASTER_HOME" ]; then
        CONTROL_DIR="$PORTMASTER_HOME"
    else
        CONTROL_DIR="$(cd "$(dirname "${BASH_SOURCE[0]:-$0}")" && pwd)"
    fi
else
    CONTROL_DIR="$controlfolder"
fi

if [ -z "$PM_VERSION" ] && [ -f "$CONTROL_DIR/version" ]; then
    export PM_VERSION="$(tr -d '\r\n' < "$CONTROL_DIR/version" 2>/dev/null)"
fi
[ -z "$PM_VERSION" ] && export PM_VERSION="Unknown"

# ------------------------------------------------------------------------------
# 1. ARCHITECTURE & MULTI-LIB COMPATIBILITY
# ------------------------------------------------------------------------------
RAW_ARCH="$(uname -m 2>/dev/null || echo "unknown")"
case "$RAW_ARCH" in
    aarch64|arm64) export DEVICE_ARCH="aarch64" ;;
    armv7*|armhf|armv8l) export DEVICE_ARCH="armhf" ;;
    x86_64|amd64)  export DEVICE_ARCH="x86_64" ;;
    i*86|x86)      export DEVICE_ARCH="x86" ;;
    *)             export DEVICE_ARCH="$RAW_ARCH" ;;
esac

export DEVICE_HAS_ARMHF="N"
export DEVICE_HAS_AARCH64="N"
export DEVICE_HAS_X86="N"
export DEVICE_HAS_X86_64="N"

case "$DEVICE_ARCH" in
    aarch64)
        export DEVICE_HAS_AARCH64="Y"
        if [ -d "/lib/arm-linux-gnueabihf" ] || [ -d "/usr/lib/arm-linux-gnueabihf" ] || [ -d "/lib32" ] || [ -d "/usr/lib32" ] || [ -f "/lib/ld-linux-armhf.so.3" ]; then
            export DEVICE_HAS_ARMHF="Y"
        fi
        ;;
    armhf)
        export DEVICE_HAS_ARMHF="Y"
        ;;
    x86_64)
        export DEVICE_HAS_X86_64="Y"
        if [ -d "/lib/i386-linux-gnu" ] || [ -d "/usr/lib/i386-linux-gnu" ] || [ -d "/usr/lib32" ] || [ -f "/lib/ld-linux.so.2" ]; then
            export DEVICE_HAS_X86="Y"
        fi
        ;;
    x86)
        export DEVICE_HAS_X86="Y"
        ;;
esac

# ------------------------------------------------------------------------------
# 2. CFW & OS DETECTION
# ------------------------------------------------------------------------------
export CFW_NAME="Unknown"
export CFW_VERSION="Unknown"

# 2a. ROCKNIX & JELOS
if [ -f "/etc/rocknix-release" ] || [ -f "/storage/.config/rocknix" ] || grep -qi 'rocknix' /etc/os-release /usr/lib/os-release 2>/dev/null; then
    export CFW_NAME="ROCKNIX"
    if [ -f "/etc/rocknix-release" ]; then
        export CFW_VERSION="$(tr -d '\r\n' < /etc/rocknix-release 2>/dev/null)"
    elif [ -f "/etc/os-release" ] || [ -f "/usr/lib/os-release" ]; then
        OS_F="/etc/os-release"; [ ! -f "$OS_F" ] && OS_F="/usr/lib/os-release"
        DATE_VER="$(grep -oE '\b20[2-3][0-9][0-1][0-9][0-3][0-9]\b' "$OS_F" 2>/dev/null | head -n1)"
        if [ -n "$DATE_VER" ]; then
            export CFW_VERSION="$DATE_VER"
        else
            for key in VERSION_ID VERSION LIBREELEC_VERSION BUILD_ID; do
                val="$(grep -iE "^${key}=" "$OS_F" 2>/dev/null | head -n1 | cut -d= -f2 | tr -d '"\r ')"
                if [ -n "$val" ]; then
                    export CFW_VERSION="$val"
                    break
                fi
            done
        fi
    fi
elif [ -f "/etc/jelos-release" ] || [ -f "/storage/.config/jelos" ] || grep -qi 'jelos' /etc/os-release /usr/lib/os-release 2>/dev/null; then
    export CFW_NAME="JELOS"
    if [ -f "/etc/jelos-release" ]; then
        export CFW_VERSION="$(tr -d '\r\n' < /etc/jelos-release 2>/dev/null)"
    elif [ -f "/etc/os-release" ] || [ -f "/usr/lib/os-release" ]; then
        OS_F="/etc/os-release"; [ ! -f "$OS_F" ] && OS_F="/usr/lib/os-release"
        DATE_VER="$(grep -oE '\b20[2-3][0-9][0-1][0-9][0-3][0-9]\b' "$OS_F" 2>/dev/null | head -n1)"
        if [ -n "$DATE_VER" ]; then
            export CFW_VERSION="$DATE_VER"
        else
            for key in VERSION_ID VERSION LIBREELEC_VERSION BUILD_ID; do
                val="$(grep -iE "^${key}=" "$OS_F" 2>/dev/null | head -n1 | cut -d= -f2 | tr -d '"\r ')"
                if [ -n "$val" ]; then
                    export CFW_VERSION="$val"
                    break
                fi
            done
        fi
    fi

# 2b. muOS (MustardOS)
elif [ -f "/opt/muos/config/system/version" ] || [ -f "/opt/muos/config/version.txt" ] || [ -d "/opt/muos" ] || grep -qiE '^(ID=muos|NAME=MustardOS)' /etc/os-release 2>/dev/null; then
    export CFW_NAME="muOS"
    if [ -f "/opt/muos/config/system/version" ]; then
        export CFW_VERSION="$(tr -d '\r\n' < /opt/muos/config/system/version 2>/dev/null)"
    elif [ -f "/opt/muos/config/version.txt" ]; then
        export CFW_VERSION="$(head -n1 /opt/muos/config/version.txt 2>/dev/null | tr -d '\r\n')"
    fi
    if [ "$CFW_VERSION" = "Unknown" ] || [ -z "$CFW_VERSION" ]; then
        [ -f "/etc/os-release" ] && export CFW_VERSION="$(grep -iE '^VERSION_ID=' /etc/os-release | cut -d= -f2 | tr -d '"\r ')"
    fi

# 2c. Plymouth Theme Matching (dArkOS, ArkOS, TheRA)
elif [ -f "/usr/share/plymouth/themes/text.plymouth" ] && grep -qiE '(darkos|arkos|thera)' /usr/share/plymouth/themes/text.plymouth 2>/dev/null; then
    PLY_TITLE="$(grep -i "title=" /usr/share/plymouth/themes/text.plymouth 2>/dev/null)"
    if echo "$PLY_TITLE" | grep -qi "darkos"; then
        export CFW_NAME="dArkOS"
    elif echo "$PLY_TITLE" | grep -qi "thera"; then
        export CFW_NAME="TheRA"
    elif echo "$PLY_TITLE" | grep -qi "arkos"; then
        export CFW_NAME="ArkOS"
    fi
    PLY_VER="$(echo "$PLY_TITLE" | grep -oE '[0-9]{6,8}' | head -n1)"
    [ -n "$PLY_VER" ] && export CFW_VERSION="$PLY_VER"

# 2d. Knulli
elif [ -f "/boot/boot/knulli.board" ] || [ -f "/boot/knulli.board" ] || [ -f "/etc/knulli.version" ] || grep -qi 'OS_NAME="*knulli"*' /etc/os-release 2>/dev/null; then
    export CFW_NAME="knulli"
    for kver in /etc/knulli.version /boot/knulli.version /userdata/system/knulli.version /etc/knulli_version; do
        if [ -f "$kver" ]; then
            export CFW_VERSION="$(tr -d '\r\n' < "$kver" 2>/dev/null)"
            break
        fi
    done
    if [ "$CFW_VERSION" = "Unknown" ] || [ -z "$CFW_VERSION" ]; then
        if [ -f "/etc/os-release" ]; then
            K_VER="$(grep -iE '^OS_VERSION=' /etc/os-release | cut -d= -f2 | tr -d '"\r ')"
            [ -n "$K_VER" ] && export CFW_VERSION="$K_VER"
        fi
    fi
    [ "$CFW_VERSION" = "Unknown" ] || [ -z "$CFW_VERSION" ] && export CFW_VERSION="scarab"

# 2e. RetroOZ
elif [ -f "/etc/retrooz_version" ]; then
    export CFW_NAME="RetroOZ"
    export CFW_VERSION="$(tr -d '\r\n' < /etc/retrooz_version 2>/dev/null)"

# 2f. AmberELEC
elif [ -f "/etc/amber_version" ] || [ -f "/etc/amberelec_version" ]; then
    export CFW_NAME="AmberELEC"
    export CFW_VERSION="$(cat /etc/amber*_version 2>/dev/null | head -n1 | tr -d '\r\n')"

# 2g. Batocera
elif [ -f "/usr/share/batocera/batocera.version" ]; then
    export CFW_NAME="Batocera.linux"
    export CFW_VERSION="$(tr -d '\r\n' < /usr/share/batocera/batocera.version 2>/dev/null)"

# 2h. Specific ArkOS/dArkOS Version File Fallbacks
elif [ -f "/etc/darkos_version" ] || [ -f "/etc/arkos_version" ]; then
    if [ -f "/etc/darkos_version" ]; then
        export CFW_NAME="dArkOS"
        export CFW_VERSION="$(tr -d '\r\n' < /etc/darkos_version 2>/dev/null)"
    else
        export CFW_NAME="ArkOS"
        export CFW_VERSION="$(tr -d '\r\n' < /etc/arkos_version 2>/dev/null)"
    fi
fi

# 2i. Generic /etc/os-release Fallback
if [ "$CFW_NAME" = "Unknown" ] && { [ -f "/etc/os-release" ] || [ -f "/usr/lib/os-release" ]; }; then
    OS_FILE="/etc/os-release"
    [ ! -f "$OS_FILE" ] && OS_FILE="/usr/lib/os-release"
    
    CFW_NAME="$(grep -E '^(NAME|ID)=' "$OS_FILE" | head -n1 | cut -d= -f2 | tr -d '"\r')"
    for key in VERSION_ID VERSION VERSION_CODENAME BUILD_ID; do
        val="$(grep -iE "^${key}=" "$OS_FILE" 2>/dev/null | head -n1 | cut -d= -f2 | tr -d '"\r ')"
        if [ -n "$val" ]; then
            export CFW_VERSION="$val"
            break
        fi
    done
fi

# 2j. GLIBC Detection (getconf -> ldd -> direct binary inspect)
export CFW_GLIBC="Unknown"
if command -v getconf >/dev/null 2>&1; then
    G_VER="$(getconf GNU_LIBC_VERSION 2>/dev/null | grep -oE '[0-9]+\.[0-9]+' | head -n1)"
    [ -n "$G_VER" ] && export CFW_GLIBC="$(echo "$G_VER" | tr -d '.')"
fi

if [ "$CFW_GLIBC" = "Unknown" ] && command -v ldd >/dev/null 2>&1; then
    L_VER="$(ldd --version 2>/dev/null | head -n1 | grep -oE '[0-9]+\.[0-9]+' | head -n1)"
    [ -n "$L_VER" ] && export CFW_GLIBC="$(echo "$L_VER" | tr -d '.')"
fi

if [ "$CFW_GLIBC" = "Unknown" ]; then
    for path in \
        /lib/libc.so.6 /lib/ld-linux*.so.* /lib64/libc.so.6 \
        /usr/lib/libc.so.6 /usr/lib64/libc.so.6 \
        /lib/aarch64-linux-gnu/libc.so.6 /usr/lib/aarch64-linux-gnu/libc.so.6 \
        /lib/arm-linux-gnueabihf/libc.so.6 /usr/lib/arm-linux-gnueabihf/libc.so.6 \
        /lib/x86_64-linux-gnu/libc.so.6 /usr/lib/x86_64-linux-gnu/libc.so.6; do
        if [ -x "$path" ] || [ -f "$path" ]; then
            GLIBC_VER="$("$path" 2>/dev/null | head -n1 | grep -oE '[0-9]+\.[0-9]+' | head -n1)"
            if [ -n "$GLIBC_VER" ]; then
                export CFW_GLIBC="$(echo "$GLIBC_VER" | tr -d '.')"
                break
            fi
        fi
    done
fi

# ------------------------------------------------------------------------------
# 3. HARDWARE (DEVICE NAME, CPU, RAM)
# ------------------------------------------------------------------------------
export DEVICE_NAME="Unknown"
export DEVICE_CPU="Unknown"

# 3a. Check CFW Board Files & Configuration Markers
for dfile in \
    "/opt/muos/device/config/board/name" \
    "/opt/muos/config/device.txt" \
    "/boot/boot/knulli.board" \
    "/boot/knulli.board" \
    "/userdata/system/knulli.board" \
    "/boot/boot/batocera.board" \
    "/boot/batocera.board" \
    "/var/run/batocera.board" \
    "/userdata/system/batocera.board" \
    "/etc/batocera.board" \
    "$HOME/.config/.CUSTOM_DEVICE" \
    "$HOME/.config/.DEVICE" \
    "$HOME/.config/.OS_ARCH" \
    "/userdata/system/.DEVICE" \
    "/userdata/system/.CUSTOM_DEVICE" \
    "/etc/device_model" \
    "/etc/board"; do
    if [ -f "$dfile" ]; then
        DEV_CONTENT="$(head -n1 "$dfile" 2>/dev/null | tr -d '\r\n')"
        if [ -n "$DEV_CONTENT" ]; then
            export DEVICE_NAME="$DEV_CONTENT"
            break
        fi
    fi
done

# 3b. Device Tree & DMI Fallback (Always null-safe)
if [ "$DEVICE_NAME" = "Unknown" ]; then
    if [ -f "/proc/device-tree/model" ]; then
        export DEVICE_NAME="$(tr -d '\0\r\n' < /proc/device-tree/model 2>/dev/null)"
    elif [ -f "/sys/firmware/devicetree/base/model" ]; then
        export DEVICE_NAME="$(tr -d '\0\r\n' < /sys/firmware/devicetree/base/model 2>/dev/null)"
    elif [ -f "/sys/devices/virtual/dmi/id/product_name" ]; then
        export DEVICE_NAME="$(tr -d '\r\n' < /sys/devices/virtual/dmi/id/product_name 2>/dev/null)"
    fi
fi

# 3c. CPU Family & SoC Resolution
if [ -f "/sys/devices/soc0/machine" ]; then
    SOC_MACH="$(tr -d '\0\r\n' < /sys/devices/soc0/machine 2>/dev/null)"
    case "$SOC_MACH" in
        QCS8550|sm8550|SM8550) export DEVICE_CPU="SM8550" ;;
        QCS8450|sm8450|SM8450) export DEVICE_CPU="SM8450" ;;
        QCS8250|sm8250|SM8250) export DEVICE_CPU="SM8250" ;;
        *) [ -n "$SOC_MACH" ] && export DEVICE_CPU="$SOC_MACH" ;;
    esac
fi

if [ "$DEVICE_CPU" = "Unknown" ]; then
    if grep -qiE '(sun50iw9|allwinner.*h700)' /proc/device-tree/compatible /proc/device-tree/model 2>/dev/null || [[ "$DEVICE_NAME" =~ ^(RG35XX|RG28XX|RG40XX|RGCUBEXX|RG34XX) ]]; then
        export DEVICE_CPU="h700"
    elif grep -qiE '(sm8550|qcom,sm8550)' /proc/device-tree/compatible /proc/device-tree/model 2>/dev/null; then
        export DEVICE_CPU="SM8550"
    elif grep -qiE '(sm8450|qcom,sm8450)' /proc/device-tree/compatible /proc/device-tree/model 2>/dev/null; then
        export DEVICE_CPU="SM8450"
    elif grep -qiE '(sm8250|qcom,sm8250)' /proc/device-tree/compatible /proc/device-tree/model 2>/dev/null; then
        export DEVICE_CPU="SM8250"
    elif grep -qiE '(rk3588|rockchip,rk3588)' /proc/device-tree/compatible /proc/device-tree/model 2>/dev/null; then
        export DEVICE_CPU="rk3588"
    elif grep -qiE '(rk3566|rockchip,rk3566)' /proc/device-tree/compatible /proc/device-tree/model 2>/dev/null; then
        export DEVICE_CPU="rk3566"
    elif grep -qiE '(rk3326|rockchip,rk3326)' /proc/device-tree/compatible /proc/device-tree/model 2>/dev/null; then
        export DEVICE_CPU="rk3326"
    fi
fi

# Fallback: ARM MIDR Part Codes Decoder
if [ "$DEVICE_CPU" = "Unknown" ] && [ -f "/proc/cpuinfo" ]; then
    PART_CODE="$(grep -i "CPU part" /proc/cpuinfo 2>/dev/null | head -n1 | cut -d: -f2 | tr -d ' \t\r\n' | tr '[:upper:]' '[:lower:]')"
    case "$PART_CODE" in
        0xd01) export DEVICE_CPU="Cortex-A32" ;;
        0xd03) export DEVICE_CPU="Cortex-A53" ;;
        0xd04) export DEVICE_CPU="Cortex-A35" ;;
        0xd05) export DEVICE_CPU="Cortex-A55" ;;
        0xd07) export DEVICE_CPU="Cortex-A57" ;;
        0xd08) export DEVICE_CPU="Cortex-A72" ;;
        0xd09) export DEVICE_CPU="Cortex-A73" ;;
        0xd0a) export DEVICE_CPU="Cortex-A75" ;;
        0xd0b) export DEVICE_CPU="Cortex-A76" ;;
        0xd0c) export DEVICE_CPU="Cortex-A77" ;;
        0xd0d) export DEVICE_CPU="Cortex-A78" ;;
        0xd46) export DEVICE_CPU="Cortex-A510" ;;
        0xd47) export DEVICE_CPU="Cortex-A710" ;;
        0xd4d) export DEVICE_CPU="Cortex-A715" ;;
        0xd4e) export DEVICE_CPU="Cortex-X3" ;;
    esac

    if [ "$DEVICE_CPU" = "Unknown" ]; then
        CPU_HW="$(grep -iE "^(Hardware|model name)" /proc/cpuinfo | head -n1 | cut -d: -f2- | sed 's/^[ \t]*//')"
        [ -n "$CPU_HW" ] && export DEVICE_CPU="$CPU_HW"
    fi
fi
[ "$DEVICE_CPU" = "Unknown" ] && export DEVICE_CPU="$RAW_ARCH"

# 3d. RAM calculation in GB (Ceiled to actual physical capacity)
export DEVICE_RAM=1
if [ -f "/proc/meminfo" ]; then
    MEM_TOTAL_KB="$(awk '/MemTotal:/ {print $2}' /proc/meminfo 2>/dev/null)"
    if [ -n "$MEM_TOTAL_KB" ] && [ "$MEM_TOTAL_KB" -gt 0 ]; then
        export DEVICE_RAM=$(( (MEM_TOTAL_KB + 1048575) / 1048576 ))
    fi
fi

# ------------------------------------------------------------------------------
# 4. DISPLAY RESOLUTION, ORIENTATION & ASPECT RATIO
# ------------------------------------------------------------------------------
export DISPLAY_WIDTH=640
export DISPLAY_HEIGHT=480
export DISPLAY_ORIENTATION=0

RES_RESOLVED=0

# DRM Modes first
if [ -d "/sys/class/drm" ]; then
    for mode_file in /sys/class/drm/card*-*/modes; do
        if [ -f "$mode_file" ]; then
            FIRST_MODE="$(head -n1 "$mode_file" 2>/dev/null)"
            if echo "$FIRST_MODE" | grep -qE '^[0-9]+x[0-9]+'; then
                export DISPLAY_WIDTH="$(echo "$FIRST_MODE" | sed -E 's/^([0-9]+)x([0-9]+).*/\1/')"
                export DISPLAY_HEIGHT="$(echo "$FIRST_MODE" | sed -E 's/^([0-9]+)x([0-9]+).*/\2/')"
                RES_RESOLVED=1
                break
            fi
        fi
    done
fi

# Framebuffer modes
if [ "$RES_RESOLVED" -eq 0 ] && [ -f "/sys/class/graphics/fb0/modes" ]; then
    FB_MODE="$(head -n1 /sys/class/graphics/fb0/modes 2>/dev/null | grep -oE '[0-9]+x[0-9]+' | head -n1)"
    if [ -n "$FB_MODE" ]; then
        export DISPLAY_WIDTH="${FB_MODE%x*}"
        export DISPLAY_HEIGHT="${FB_MODE#*x}"
        RES_RESOLVED=1
    fi
fi

# Framebuffer virtual size fallback
if [ "$RES_RESOLVED" -eq 0 ] && [ -f "/sys/class/graphics/fb0/virtual_size" ]; then
    FB_SIZE="$(cat /sys/class/graphics/fb0/virtual_size 2>/dev/null)"
    FB_W="${FB_SIZE%,*}"
    FB_H="${FB_SIZE#*,}"
    if [ "$FB_W" -gt 0 ] 2>/dev/null && [ "$FB_H" -gt 0 ] 2>/dev/null; then
        export DISPLAY_WIDTH="$FB_W"
        export DISPLAY_HEIGHT="$FB_H"
    fi
fi

# Binary helper fallback
if [ -x "$CONTROL_DIR/sdl_resolution" ]; then
    RES_OUT="$("$CONTROL_DIR/sdl_resolution" 2>/dev/null)"
    if [ -n "$RES_OUT" ]; then
        export DISPLAY_WIDTH="$(echo "$RES_OUT" | awk '{print $1}')"
        export DISPLAY_HEIGHT="$(echo "$RES_OUT" | awk '{print $2}')"
    fi
fi

# Clean integer values
DISPLAY_WIDTH="$(echo "$DISPLAY_WIDTH" | grep -oE '^[0-9]+' || echo "640")"
DISPLAY_HEIGHT="$(echo "$DISPLAY_HEIGHT" | grep -oE '^[0-9]+' || echo "480")"

# Orientation check
if [ -f "/sys/class/graphics/fb0/rotate" ]; then
    case "$(cat /sys/class/graphics/fb0/rotate 2>/dev/null)" in
        1) export DISPLAY_ORIENTATION=90 ;;
        2) export DISPLAY_ORIENTATION=180 ;;
        3) export DISPLAY_ORIENTATION=270 ;;
        *) export DISPLAY_ORIENTATION=0 ;;
    esac
fi

# Dynamic GCD Aspect Ratio calculation
calc_gcd() {
    local a=$1 b=$2
    while [ "$b" -ne 0 ]; do
        local t=$b
        b=$((a % b))
        a=$t
    done
    echo "$a"
}

GCD=$(calc_gcd "$DISPLAY_WIDTH" "$DISPLAY_HEIGHT")
if [ "$GCD" -gt 0 ] 2>/dev/null; then
    export ASPECT_X=$(( DISPLAY_WIDTH / GCD ))
    export ASPECT_Y=$(( DISPLAY_HEIGHT / GCD ))
    if [ "$ASPECT_X" -eq 8 ] && [ "$ASPECT_Y" -eq 5 ]; then
        export ASPECT_X=16
        export ASPECT_Y=10
    fi
else
    export ASPECT_X=4
    export ASPECT_Y=3
fi

# ------------------------------------------------------------------------------
# 5. CONTROLS (ANALOG STICKS)
# ------------------------------------------------------------------------------
export ANALOG_STICKS=2

case "$DEVICE_NAME" in
    *"RG35XX-PLUS"*|*"RG35XX-SP"*|*"RG28XX"*|*"RG35XX-2024"*|*"RG34XX"*|*"MIYOO MINI"*)
        export ANALOG_STICKS=0
        ;;
    *"RG351V"*|*"RGB20S"*|*"RG40XX-V"*)
        export ANALOG_STICKS=1
        ;;
esac

# ------------------------------------------------------------------------------
# 6. OUTPUT & FILE GENERATION
# ------------------------------------------------------------------------------
save_device_info() {
    local target_file="$1"
    cat <<EOF > "$target_file"
DEVICE_INFO_VERSION="$DEVICE_INFO_VERSION"
PM_VERSION="$PM_VERSION"
CFW_NAME="$CFW_NAME"
CFW_VERSION="$CFW_VERSION"
CFW_GLIBC="$CFW_GLIBC"
DEVICE_NAME="$DEVICE_NAME"
DEVICE_CPU="$DEVICE_CPU"
DEVICE_ARCH="$DEVICE_ARCH"
DEVICE_RAM="$DEVICE_RAM"
DEVICE_HAS_ARMHF="$DEVICE_HAS_ARMHF"
DEVICE_HAS_AARCH64="$DEVICE_HAS_AARCH64"
DEVICE_HAS_X86="$DEVICE_HAS_X86"
DEVICE_HAS_X86_64="$DEVICE_HAS_X86_64"
DISPLAY_WIDTH="$DISPLAY_WIDTH"
DISPLAY_HEIGHT="$DISPLAY_HEIGHT"
DISPLAY_ORIENTATION="$DISPLAY_ORIENTATION"
ASPECT_X="$ASPECT_X"
ASPECT_Y="$ASPECT_Y"
ANALOG_STICKS="$ANALOG_STICKS"
EOF
}

if [ -d "$CONTROL_DIR" ] && [ -w "$CONTROL_DIR" ]; then
    save_device_info "$CONTROL_DIR/device_info.env"
fi

if [ "$SOURCED" -eq 0 ]; then
    echo "========================================"
    echo " PortMaster Device Information (v$DEVICE_INFO_VERSION)"
    echo "========================================"
    printf "%-20s: %s\n" "CFW_NAME" "$CFW_NAME"
    printf "%-20s: %s\n" "CFW_VERSION" "$CFW_VERSION"
    printf "%-20s: %s\n" "CFW_GLIBC" "$CFW_GLIBC"
    printf "%-20s: %s\n" "DEVICE_NAME" "$DEVICE_NAME"
    printf "%-20s: %s\n" "DEVICE_CPU" "$DEVICE_CPU"
    printf "%-20s: %s\n" "DEVICE_ARCH" "$DEVICE_ARCH"
    printf "%-20s: %s GB\n" "DEVICE_RAM" "$DEVICE_RAM"
    printf "%-20s: %s (ARMHF:%s, AARCH64:%s)\n" "CAPABILITIES" "$DEVICE_ARCH" "$DEVICE_HAS_ARMHF" "$DEVICE_HAS_AARCH64"
    printf "%-20s: %sx%s (%s:%s)\n" "DISPLAY" "$DISPLAY_WIDTH" "$DISPLAY_HEIGHT" "$ASPECT_X" "$ASPECT_Y"
    printf "%-20s: %s\n" "ORIENTATION" "$DISPLAY_ORIENTATION"
    printf "%-20s: %s\n" "ANALOG_STICKS" "$ANALOG_STICKS"
    echo "========================================"
fi
