#!/bin/bash
# ==============================================================================
# PortMaster Device Information Provider (Dynamic Modernized v0.2.0)
# ==============================================================================

(return 0 2>/dev/null) && SOURCED=1 || SOURCED=0

if [ "$SOURCED" -eq 1 ] && [ -n "$DEVICE_INFO_VERSION" ]; then
    return 0 2>/dev/null
fi

export DEVICE_INFO_VERSION="0.2.0"

# ------------------------------------------------------------------------------
# 0. DIRECTORY SETUP
# ------------------------------------------------------------------------------
if [ -z "$controlfolder" ]; then
    if [ -n "$PORTMASTER_HOME" ] && [ -d "$PORTMASTER_HOME" ]; then
        CONTROL_DIR="$PORTMASTER_HOME"
    else
        CONTROL_DIR="$(cd "$(dirname "${BASH_SOURCE[0]:-$0}")" && pwd)"
    fi
else
    CONTROL_DIR="$controlfolder"
fi

if [ -z "$PM_VERSION" ] && [ -f "$CONTROL_DIR/version" ]; then
    export PM_VERSION="$(tr -d '\r\n' < "$CONTROL_DIR/version" 2>/dev/null)"
fi
[ -z "$PM_VERSION" ] && export PM_VERSION="Unknown"

# ------------------------------------------------------------------------------
# 1. ARCHITECTURE & MULTI-LIB COMPATIBILITY
# ------------------------------------------------------------------------------
RAW_ARCH="$(uname -m 2>/dev/null || echo "unknown")"
case "$RAW_ARCH" in
    aarch64|arm64) export DEVICE_ARCH="aarch64" ;;
    armv7*|armhf|armv8l) export DEVICE_ARCH="armhf" ;;
    x86_64|amd64)  export DEVICE_ARCH="x86_64" ;;
    i*86|x86)      export DEVICE_ARCH="x86" ;;
    *)             export DEVICE_ARCH="$RAW_ARCH" ;;
esac

export DEVICE_HAS_ARMHF="N"
export DEVICE_HAS_AARCH64="N"
export DEVICE_HAS_X86="N"
export DEVICE_HAS_X86_64="N"

case "$DEVICE_ARCH" in
    aarch64)
        export DEVICE_HAS_AARCH64="Y"
        if [ -d "/lib/arm-linux-gnueabihf" ] || [ -d "/usr/lib/arm-linux-gnueabihf" ] || [ -d "/lib32" ] || [ -d "/usr/lib32" ] || [ -f "/lib/ld-linux-armhf.so.3" ]; then
            export DEVICE_HAS_ARMHF="Y"
        fi
        ;;
    armhf)
        export DEVICE_HAS_ARMHF="Y"
        ;;
    x86_64)
        export DEVICE_HAS_X86_64="Y"
        if [ -d "/lib/i386-linux-gnu" ] || [ -d "/usr/lib/i386-linux-gnu" ] || [ -d "/usr/lib32" ] || [ -f "/lib/ld-linux.so.2" ]; then
            export DEVICE_HAS_X86="Y"
        fi
        ;;
    x86)
        export DEVICE_HAS_X86="Y"
        ;;
esac

# ------------------------------------------------------------------------------
# 2. CFW & OS DETECTION
# ------------------------------------------------------------------------------
export CFW_NAME="Unknown"
export CFW_VERSION="Unknown"

# 2a. RetroDECK Flatpak Container
if [ -f "/app/bin/retrodeck" ] || [ -f "/app/bin/retrodeck.sh" ] || [ "$FLATPAK_ID" = "net.retrodeck.retrodeck" ]; then
    export CFW_NAME="RetroDECK"
    if [ -f "/var/config/retrodeck/retrodeck.json" ] && command -v jq >/dev/null 2>&1; then
        export CFW_VERSION="$(jq -r .version /var/config/retrodeck/retrodeck.json 2>/dev/null)"
    elif [ -f "/var/config/retrodeck/retrodeck.cfg" ]; then
        export CFW_VERSION="$(grep -i 'version' /var/config/retrodeck/retrodeck.cfg 2>/dev/null | cut -d= -f2 | tr -d '"\r ')"
    fi

# 2b. ROCKNIX & JELOS
elif [ -f "/etc/rocknix-release" ] || [ -f "/storage/.config/rocknix" ] || grep -qi 'rocknix' /etc/os-release /usr/lib/os-release 2>/dev/null; then
    export CFW_NAME="ROCKNIX"
    if [ -f "/etc/rocknix-release" ]; then
        export CFW_VERSION="$(tr -d '\r\n' < /etc/rocknix-release 2>/dev/null)"
    elif [ -f "/etc/os-release" ] || [ -f "/usr/lib/os-release" ]; then
        OS_F="/etc/os-release"; [ ! -f "$OS_F" ] && OS_F="/usr/lib/os-release"
        DATE_VER="$(grep -oE '\b20[2-3][0-9][0-1][0-9][0-3][0-9]\b' "$OS_F" 2>/dev/null | head -n1)"
        if [ -n "$DATE_VER" ]; then
            export CFW_VERSION="$DATE_VER"
        else
            for key in VERSION_ID VERSION LIBREELEC_VERSION BUILD_ID; do
                val="$(grep -iE "^${key}=" "$OS_F" 2>/dev/null | head -n1 | cut -d= -f2 | tr -d '"\r ')"
                if [ -n "$val" ]; then
                    export CFW_VERSION="$val"
                    break
                fi
            done
        fi
    fi
elif [ -f "/etc/jelos-release" ] || [ -f "/storage/.config/jelos" ] || grep -qi 'jelos' /etc/os-release /usr/lib/os-release 2>/dev/null; then
    export CFW_NAME="JELOS"
    if [ -f "/etc/jelos-release" ]; then
        export CFW_VERSION="$(tr -d '\r\n' < /etc/jelos-release 2>/dev/null)"
    elif [ -f "/etc/os-release" ] || [ -f "/usr/lib/os-release" ]; then
        OS_F="/etc/os-release"; [ ! -f "$OS_F" ] && OS_F="/usr/lib/os-release"
        DATE_VER="$(grep -oE '\b20[2-3][0-9][0-1][0-9][0-3][0-9]\b' "$OS_F" 2>/dev/null | head -n1)"
        if [ -n "$DATE_VER" ]; then
            export CFW_VERSION="$DATE_VER"
        else
            for key in VERSION_ID VERSION LIBREELEC_VERSION BUILD_ID; do
                val="$(grep -iE "^${key}=" "$OS_F" 2>/dev/null | head -n1 | cut -d= -f2 | tr -d '"\r ')"
                if [ -n "$val" ]; then
                    export CFW_VERSION="$val"
                    break
                fi
            done
        fi
    fi

# 2c. muOS (MustardOS)
elif [ -f "/opt/muos/config/system/version" ] || [ -f "/opt/muos/config/version.txt" ] || [ -d "/opt/muos" ] || grep -qiE '^(ID=muos|NAME=MustardOS)' /etc/os-release 2>/dev/null; then
    export CFW_NAME="muOS"
    if [ -f "/opt/muos/config/system/version" ]; then
        export CFW_VERSION="$(tr -d '\r\n' < /opt/muos/config/system/version 2>/dev/null)"
    elif [ -f "/opt/muos/config/version.txt" ]; then
        export CFW_VERSION="$(head -n1 /opt/muos/config/version.txt 2>/dev/null | tr -d '\r\n')"
    fi
    if [ "$CFW_VERSION" = "Unknown" ] || [ -z "$CFW_VERSION" ]; then

    fi

# 2d. Spruce

    CFW_NAME="spruce"

# 2e. Plymouth Theme (dArkOS, ArkOS, TheRA)

    PLY_TITLE="$(grep -i "title=" /usr/share/plymouth/themes/text.plymouth 2>/dev/null)"
    if echo "$PLY_TITLE" | grep -qi "darkos"; then
        export CFW_NAME="dArkOS"
    elif echo "$PLY_TITLE" | grep -qi "thera"; then
        export CFW_NAME="TheRA"
    elif echo "$PLY_TITLE" | grep -qi "arkos"; then
        export CFW_NAME="ArkOS"
    fi
    PLY_VER="$(echo "$PLY_TITLE" | grep -oE '[0-9]{6,8}' | head -n1)"
    [ -n "$PLY_VER" ] && export CFW_VERSION="$PLY_VER"

# 2f. Knulli

        if [ -f "$kver" ]; then

            break
        fi
    done
    if [ "$CFW_VERSION" = "Unknown" ] || [ -z "$CFW_VERSION" ]; then
        if [ -f "/etc/os-release" ]; then
            K_VER="$(grep -iE '^OS_VERSION=' /etc/os-release | cut -d= -f2 | tr -d '"\r ')"

        fi

# 2g. RetroOZ
elif [ -f "/etc/retrooz_version" ]; then
    export CFW_NAME="RetroOZ"
    export CFW_VERSION="$(tr -d '\r\n' < /etc/retrooz_version 2>/dev/null)"

# 2h. AmberELEC
elif [ -f "/etc/amber_version" ] || [ -f "/etc/amberelec_version" ]; then

# 2i. Batocera

    export CFW_VERSION="$(tr -d '\r\n' < /usr/share/batocera/batocera.version 2>/dev/null)"

# 2j. Specific ArkOS/dArkOS Version File Fallbacks
elif [ -f "/etc/darkos_version" ] || [ -f "/etc/arkos_version" ]; then
    if [ -f "/etc/darkos_version" ]; then
        export CFW_NAME="dArkOS"
        export CFW_VERSION="$(tr -d '\r\n' < /etc/darkos_version 2>/dev/null)"
    else
        export CFW_NAME="ArkOS"
        export CFW_VERSION="$(tr -d '\r\n' < /etc/arkos_version 2>/dev/null)"
    fi
fi

# 2k. Generic /etc/os-release Fallback (SteamOS, Debian, Ubuntu, Fedora, Arch)
if [ "$CFW_NAME" = "Unknown" ] && { [ -f "/etc/os-release" ] || [ -f "/usr/lib/os-release" ]; }; then
    OS_FILE="/etc/os-release"; [ ! -f "$OS_FILE" ] && OS_FILE="/usr/lib/os-release"
    CFW_NAME="$(grep -E '^(NAME|ID)=' "$OS_FILE" | head -n1 | cut -d= -f2 | tr -d '"\r')"
    for key in VERSION_ID VERSION VERSION_CODENAME BUILD_ID; do
        val="$(grep -iE "^${key}=" "$OS_FILE" 2>/dev/null | head -n1 | cut -d= -f2 | tr -d '"\r ')"
        if [ -n "$val" ]; then
            export CFW_VERSION="$val"

fi

# 2l. GLIBC Detection
export CFW_GLIBC="Unknown"
if command -v getconf >/dev/null 2>&1; then

        if [ -x "$path" ] || [ -f "$path" ]; then
            GLIBC_VER="$("$path" 2>/dev/null | head -n1 | grep -oE '[0-9]+\.[0-9]+' | head -n1)"

                export CFW_GLIBC="$(echo "$GLIBC_VER" | tr -d '.')"

# 3c. CPU Family & SoC Resolution
detect_dynamic_cpu() {
    # 1. Linux generic SoC bus (Snapdragon, Exynos, etc.)
    for soc_dir in /sys/devices/soc0 /sys/bus/soc/devices/soc0; do
        if [ -f "$soc_dir/machine" ]; then
            local soc_mach="$(tr -d '\0\r\n' < "$soc_dir/machine" 2>/dev/null)"
            if [ -n "$soc_mach" ]; then echo "$soc_mach"; return; fi
        fi
    done

    # 2. Positional Devicetree SoC extraction
    local dt_comp="/proc/device-tree/compatible"
    [ ! -f "$dt_comp" ] && dt_comp="/sys/firmware/devicetree/base/compatible"
    if [ -f "$dt_comp" ]; then
        local soc_line="$(tr '\0' '\n' < "$dt_comp" 2>/dev/null | sed -n '2p')"
        [ -z "$soc_line" ] && soc_line="$(tr '\0' '\n' < "$dt_comp" 2>/dev/null | head -n1)"
        if [ -n "$soc_line" ]; then
            local clean_soc="${soc_line#*,}"
            clean_soc="${clean_soc#sun*-}"
            # Normalize Allwinner BSP kernel silicon codes to standard PortMaster names
            case "$clean_soc" in
                sun50iw9*)  clean_soc="h700" ;;
                sun50iw10*) clean_soc="a133plus" ;;
                sun50iw12*) clean_soc="a527" ;;
                sun50iw6*)  clean_soc="h6" ;;
            esac
            if [ -n "$clean_soc" ]; then echo "$clean_soc"; return; fi
        fi
    fi

    # 3. /proc/cpuinfo Hardware or model name fallback
    if [ -f "/proc/cpuinfo" ]; then
        local cpu_hw="$(grep -iE '^(model name|Hardware)' /proc/cpuinfo 2>/dev/null | head -n1 | cut -d: -f2- | sed 's/^[ \t]*//' | tr -d '\r\n')"
        if [ -n "$cpu_hw" ] && [ "$cpu_hw" != "Unknown" ] && [ "$cpu_hw" != "dummy" ]; then
            echo "$cpu_hw"; return
        fi
    fi

    uname -m 2>/dev/null || echo "Unknown"
}
export DEVICE_CPU="$(detect_dynamic_cpu)"

# 3d. RAM calculation in GB (Ceiled to actual physical capacity)

    MEM_TOTAL_KB="$(awk '/MemTotal:/ {print $2}' /proc/meminfo 2>/dev/null)"
    if [ -n "$MEM_TOTAL_KB" ] && [ "$MEM_TOTAL_KB" -gt 0 ]; then
        export DEVICE_RAM=$(( (MEM_TOTAL_KB + 1048575) / 1048576 ))
    fi

DISPLAY_HEIGHT="$(echo "$DISPLAY_HEIGHT" | grep -oE '^[0-9]+' || echo "480")"

# Normalize native portrait handheld panels (e.g. Steam Deck 800x1280 -> 1280x800)

    if [ "$DISPLAY_WIDTH" -lt "$DISPLAY_HEIGHT" ]; then
        TMP_W="$DISPLAY_WIDTH"
        export DISPLAY_WIDTH="$DISPLAY_HEIGHT"
        export DISPLAY_HEIGHT="$TMP_W"

        1) export DISPLAY_ORIENTATION=90 ;;

        3) export DISPLAY_ORIENTATION=270 ;;
        *) export DISPLAY_ORIENTATION=0 ;;

# Dynamic GCD Aspect Ratio calculation
calc_gcd() {
    local a=$1 b=$2
    while [ "$b" -ne 0 ]; do
        local t=$b
        b=$((a % b))
        a=$t
    done
    echo "$a"
}

GCD=$(calc_gcd "$DISPLAY_WIDTH" "$DISPLAY_HEIGHT")
if [ "$GCD" -gt 0 ] 2>/dev/null; then
    export ASPECT_X=$(( DISPLAY_WIDTH / GCD ))
    export ASPECT_Y=$(( DISPLAY_HEIGHT / GCD ))
    if [ "$ASPECT_X" -eq 8 ] && [ "$ASPECT_Y" -eq 5 ]; then
        export ASPECT_X=16
        export ASPECT_Y=10
    fi
else
    export ASPECT_X=4
    export ASPECT_Y=3
fi

# ------------------------------------------------------------------------------
# 5. CONTROLS (ANALOG STICKS)
# ------------------------------------------------------------------------------
detect_dynamic_controls() {
    local max_sticks=0
    local has_triggers="N"

    for abs_file in /sys/class/input/input*/capabilities/abs; do
        [ -f "$abs_file" ] || continue
        
        local dev_name="$(cat "${abs_file%/*}/name" 2>/dev/null | tr '[:upper:]' '[:lower:]')"
        case "$dev_name" in
            *touch*|*stylus*|*accel*|*gyro*|*sensor*|*lid*|*power*|*sleep*) continue ;;
        esac

        local raw_abs="$(cat "$abs_file" 2>/dev/null)"
        [ -z "$raw_abs" ] && continue

        local lowest_word="${raw_abs##* }"
        lowest_word="${lowest_word##*,}"
        local val=$(( 0x${lowest_word} )) 2>/dev/null || continue

        local current_sticks=0
        # Stick 1: ABS_X (bit 0) & ABS_Y (bit 1) -> 0x3
        if [ $(( val & 0x3 )) -eq 3 ]; then
            current_sticks=1
            # Stick 2: ABS_RX (bit 3) & ABS_RY (bit 4) -> 0x18 OR ABS_Z (bit 2) & ABS_RZ (bit 5) -> 0x24
            if [ $(( val & 0x18 )) -eq 24 ] || [ $(( val & 0x24 )) -eq 36 ]; then
                current_sticks=2
            fi
        fi
        [ "$current_sticks" -gt "$max_sticks" ] && max_sticks="$current_sticks"

        # Analog Triggers:
        # Requires ABS_GAS + ABS_BRAKE (0x600) OR all 6 axes X+Y+RX+RY+Z+RZ (0x3f / 63)
        # (Avoids false positive when Z/RZ acts as Stick 2 without RX/RY)
        if [ $(( val & 0x600 )) -eq 1536 ] || [ $(( val & 0x3f )) -eq 63 ]; then
            has_triggers="Y"
        fi
    done

    # Fallback for GPIO-only buttons
    if [ "$max_sticks" -eq 0 ]; then
        local u_name="$(echo "$DEVICE_NAME" | tr '[:lower:]' '[:upper:]')"
        case "$u_name" in
            *"RG35XX-PLUS"*|*"RG35XX-SP"*|*"RG28XX"*|*"RG35XX-2024"*|*"RG34XX"*|*"MIYOO MINI"*) max_sticks=0 ;;
            *"RG351V"*|*"RGB20S"*|*"RG40XX-V"*) max_sticks=1 ;;
            *) max_sticks=2 ;;
        esac
    fi

    export ANALOG_STICKS="$max_sticks"
    export ANALOG_TRIGGERS="$has_triggers"
}

detect_dynamic_controls

# ------------------------------------------------------------------------------

    local target_file="$1"
    cat <<EOF > "$target_file"

CFW_GLIBC="$CFW_GLIBC"

DISPLAY_WIDTH="$DISPLAY_WIDTH"
DISPLAY_HEIGHT="$DISPLAY_HEIGHT"
DISPLAY_ORIENTATION="$DISPLAY_ORIENTATION"
ASPECT_X="$ASPECT_X"
ASPECT_Y="$ASPECT_Y"
